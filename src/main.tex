% ------------------------------------------------- Mehrspaltiger Block
% \begin{mehrspaltig}{3,2} ... \spaltenwechsel ... \end{mehrspaltig}
% Feste Spaltenbreiten; Inhalte per \spaltenwechsel manuell verteilen.
% Das Argument sind die Anteile der Spalten (beliebige Zahlen, beliebig viele):
%   {1,1}   zwei gleich breite Spalten
%   {3,2}   linke Spalte 60 %, rechte 40 %
%   {1,2,1} drei Spalten, die mittlere doppelt so breit
\ExplSyntaxOn
\NewDocumentEnvironment{mehrspaltig}{m}
  {
    \clist_set:Nn \l__bb_parts_clist {#1}
    \int_set:Nn \l__bb_n_int { \clist_count:N \l__bb_parts_clist }
    \fp_set:Nn \l__bb_sum_fp { 0 }
    \clist_map_inline:Nn \l__bb_parts_clist
      { \fp_add:Nn \l__bb_sum_fp { ##1 } }
    \clist_clear:N \l__bb_ratio_clist
    \int_zero:N \l__bb_i_int
    \clist_map_inline:Nn \l__bb_parts_clist
      {
        \int_incr:N \l__bb_i_int
        \int_compare:nT { \l__bb_i_int < \l__bb_n_int }
          {
            \clist_put_right:Nx \l__bb_ratio_clist
              { \fp_eval:n { round( ##1 / \l__bb_sum_fp , 4 ) } }
          }
      }
    \exp_args:NV \columnratio \l__bb_ratio_clist
    \use:x { \exp_not:N \begin { paracol } { \int_use:N \l__bb_n_int } }
  }
  { \end{paracol} }
\clist_new:N \l__bb_parts_clist
\clist_new:N \l__bb_ratio_clist
\fp_new:N \l__bb_sum_fp
\int_new:N \l__bb_n_int
\int_new:N \l__bb_i_int
\ExplSyntaxOff
